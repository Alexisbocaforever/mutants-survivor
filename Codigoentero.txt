#ifndef ARMA_H
#define ARMA_H

#include "dibujable_abstracto.h"
#include "proyectil.h"
#include <QPointF>
#include <QList>

class Arma : public Dibujable_abstracto
{
public:
    Arma();
    virtual ~Arma() = default;

    virtual void actualizar(float posX, float posY, QPointF mouseWorld) = 0;
    virtual void disparar() = 0;
    virtual void actualizarProyectiles() = 0;
    virtual QList<Proyectil*>& getProyectiles() = 0;

    void setAngulo(float angulo);
    float getAngulo() const;

    // Sistema de recarga
    void setMultiplicadorRecarga(float multiplicador);
    bool estaRecargando() const;
    int getMunicionActual() const;
    int getMunicionMaxima() const;
    float getProgresoRecarga() const; // 0.0 a 1.0

protected:
    float pos_x;
    float pos_y;
    float angulo;
    float ancho;
    float alto;

    // Sistema de munición y recarga
    int municionActual;
    int municionMaxima;
    float tiempoRecarga;
    float tiempoRecargaMax;
    float multiplicadorRecarga;
    bool recargando;

    void iniciarRecarga();
    void actualizarRecarga();
};

#endif // ARMA_H
#include "arma.h"

Arma::Arma() : pos_x(0), pos_y(0), angulo(0), ancho(0), alto(0),
    municionActual(0), municionMaxima(10), tiempoRecarga(0.0f),
    tiempoRecargaMax(2.0f), multiplicadorRecarga(1.0f), recargando(false)
{
}

void Arma::setAngulo(float angulo)
{
    this->angulo = angulo;
}

float Arma::getAngulo() const
{
    return angulo;
}

void Arma::setMultiplicadorRecarga(float multiplicador)
{
    multiplicadorRecarga = multiplicador;
}

bool Arma::estaRecargando() const
{
    return recargando;
}

int Arma::getMunicionActual() const
{
    return municionActual;
}

int Arma::getMunicionMaxima() const
{
    return municionMaxima;
}

float Arma::getProgresoRecarga() const
{
    if (!recargando || tiempoRecargaMax <= 0) return 1.0f;
    return 1.0f - (tiempoRecarga / tiempoRecargaMax);
}

void Arma::iniciarRecarga()
{
    if (recargando) return;
    recargando = true;
    tiempoRecarga = tiempoRecargaMax * multiplicadorRecarga;
}

void Arma::actualizarRecarga()
{
    if (!recargando) return;

    tiempoRecarga -= 0.016f; // ~16ms por frame

    if (tiempoRecarga <= 0) {
        tiempoRecarga = 0;
        recargando = false;
        municionActual = municionMaxima;
    }
}

    float get_alto() override;

BaseMilitar::BaseMilitar()
{
    // Base militar centrada en el mundo
    x_escenario = 600.0f;
    y_escenario = 400.0f;
    ancho_escenario = 400.0f;
    alto_escenario = 400.0f;

    crearValla();
}

void BaseMilitar::crearValla()
{
    float grosor = 10.0f;
    float margen = 5.0f; // Espacio entre paredes dobles

    // Limpiar paredes anteriores
    qDeleteAll(paredes);
    paredes.clear();

    // VALLA PERIMETRAL DOBLE (más realista)

    // Pared NORTE (arriba) - doble
    agregarPared(new Pared(get_x() - grosor, get_y() - grosor,
                           get_ancho() + 2*grosor, grosor));


    // Pared SUR (abajo) - doble
    agregarPared(new Pared(get_x() - grosor, get_y() + get_alto(),
                           get_ancho() + 2*grosor, grosor));


    // Pared OESTE (izquierda) - doble
    agregarPared(new Pared(get_x() - grosor, get_y() - grosor,
                           grosor, get_alto() + 2*grosor));


    // Pared ESTE (derecha) - doble
    agregarPared(new Pared(get_x() + get_ancho(), get_y() - grosor,
                           grosor, get_alto() + 2*grosor));





}

void BaseMilitar::dibujar(QPainter &p)
{
    // Dibujar suelo de la base (cemento)
    p.save();

    QColor colorSuelo(120, 120, 125);
    p.setBrush(colorSuelo);
    p.setPen(Qt::NoPen);
    p.drawRect(QRectF(get_x(), get_y(), get_ancho(), get_alto()));


    p.restore();

    // Dibujar valla y torres
    dibujarParedes(p);

    // Dibujar cartel
    dibujarCartel(p);
}

void BaseMilitar::dibujarCartel(QPainter &p)
{
    // Cartel arriba de la entrada (centro-sur)
    const float cartelX = get_x() + get_ancho() / 2.0f - 150.0f;
    const float cartelY = get_y() - 80.0f;
    const float cartelAncho = 300.0f;
    const float cartelAlto = 60.0f;

    p.save();
    p.setRenderHint(QPainter::Antialiasing, true);

    // Dimensiones relativas
    const float posteW = 10.0f;
    const float posteH = 30.0f;
    const float posteOffset = 20.0f;
    const QRectF cartelRect(cartelX, cartelY, cartelAncho, cartelAlto);

    // Postes del cartel (con degradado y "remaches")
    QLinearGradient gradPoste(cartelX, cartelY + cartelAlto, cartelX + posteW, cartelY + cartelAlto + posteH);
    gradPoste.setColorAt(0.0, QColor(70, 70, 70));
    gradPoste.setColorAt(1.0, QColor(40, 40, 40));
    p.setBrush(gradPoste);
    p.setPen(Qt::NoPen);
    p.drawRect(QRectF(cartelX + posteOffset, cartelY + cartelAlto, posteW, posteH));
    p.drawRect(QRectF(cartelX + cartelAncho - posteOffset - posteW, cartelY + cartelAlto, posteW, posteH));

    // Remaches en postes
    p.setBrush(QColor(180, 180, 180));
    p.drawEllipse(QPointF(cartelX + posteOffset + posteW/2.0f, cartelY + cartelAlto + 6.0f), 2.5f, 2.5f);
    p.drawEllipse(QPointF(cartelX + posteOffset + posteW/2.0f, cartelY + cartelAlto + 18.0f), 2.5f, 2.5f);
    p.drawEllipse(QPointF(cartelX + cartelAncho - posteOffset - posteW/2.0f, cartelY + cartelAlto + 6.0f), 2.5f, 2.5f);
    p.drawEllipse(QPointF(cartelX + cartelAncho - posteOffset - posteW/2.0f, cartelY + cartelAlto + 18.0f), 2.5f, 2.5f);

    // Fondo del cartel (degradado militar)
    QLinearGradient gradFondo(cartelRect.topLeft(), cartelRect.bottomLeft());
    gradFondo.setColorAt(0.0, QColor(95, 115, 85));   // más claro arriba
    gradFondo.setColorAt(1.0, QColor(70, 90, 60));    // más oscuro abajo
    p.setBrush(gradFondo);
    p.setPen(Qt::NoPen);
    p.drawRoundedRect(cartelRect, 6.0f, 6.0f);

    // Borde exterior
    p.setBrush(Qt::NoBrush);
    p.setPen(QPen(QColor(50, 60, 45), 3));
    p.drawRoundedRect(cartelRect, 6.0f, 6.0f);

    // Banda interior superior (para dar jerarquía)
    QRectF banda(cartelX + 6.0f, cartelY + 6.0f, cartelAncho - 12.0f, cartelAlto * 0.28f);
    QLinearGradient gradBanda(banda.topLeft(), banda.bottomLeft());
    gradBanda.setColorAt(0.0, QColor(110, 130, 100, 200));
    gradBanda.setColorAt(1.0, QColor(90, 110, 80, 200));
    p.setBrush(gradBanda);
    p.setPen(Qt::NoPen);
    p.drawRoundedRect(banda, 4.0f, 4.0f);

    // Emblema / insignia a la izquierda (círculo con estrella)
    const float emblemaSize = cartelAlto - 14.0f;
    const QRectF emblemaRect(cartelX + 10.0f, cartelY + (cartelAlto - emblemaSize) / 2.0f, emblemaSize, emblemaSize);
    // Círculo exterior
    QRadialGradient gradEmblema(emblemaRect.center(), emblemaSize/2.0f);
    gradEmblema.setColorAt(0.0, QColor(220, 220, 180));
    gradEmblema.setColorAt(1.0, QColor(140, 140, 100));
    p.setBrush(gradEmblema);
    p.setPen(QPen(QColor(60, 60, 40), 1));
    p.drawEllipse(emblemaRect);

    // Estrella simple en el centro
    QPainterPath estrella;
    const QPointF c = emblemaRect.center();
    const float R = emblemaSize * 0.38f;
    const float r = R * 0.45f;
    for (int i = 0; i < 5; ++i) {
        const double ang = (i * 72.0 - 90.0) * M_PI / 180.0;
        const double ang2 = (i * 72.0 + 36.0 - 90.0) * M_PI / 180.0;
        QPointF p1(c.x() + R * cos(ang), c.y() + R * sin(ang));
        QPointF p2(c.x() + r * cos(ang2), c.y() + r * sin(ang2));
        if (i == 0) estrella.moveTo(p1); else estrella.lineTo(p1);
        estrella.lineTo(p2);
    }
    estrella.closeSubpath();
    p.setBrush(QColor(245, 245, 200));
    p.setPen(Qt::NoPen);
    p.drawPath(estrella);

    // Bandera argentina en el espacio derecho del cartel
    // Colocamos la bandera dentro de un rect pequeño a la derecha, alineada verticalmente al centro del cartel
    const float flagW = emblemaSize * 1.2f;
    const float flagH = emblemaSize * 0.6f;
    const QRectF flagRect(cartelX + cartelAncho - 10.0f - flagW, cartelY + (cartelAlto - flagH) / 2.0f, flagW, flagH);

    // Tres franjas horizontales: celeste - blanco - celeste
    QColor celeste(116, 172, 223); // celeste bandera
    QRectF topStripe(flagRect.left(), flagRect.top(), flagRect.width(), flagRect.height() / 3.0f);
    QRectF midStripe(flagRect.left(), flagRect.top() + flagRect.height() / 3.0f, flagRect.width(), flagRect.height() / 3.0f);
    QRectF botStripe(flagRect.left(), flagRect.top() + 2.0f * flagRect.height() / 3.0f, flagRect.width(), flagRect.height() / 3.0f);

    p.setPen(Qt::NoPen);
    p.setBrush(celeste);
    p.drawRect(topStripe);
    p.setBrush(Qt::white);
    p.drawRect(midStripe);
    p.setBrush(celeste);
    p.drawRect(botStripe);

    // Borde de la bandera
    p.setPen(QPen(QColor(40, 40, 40), 1));
    p.setBrush(Qt::NoBrush);
    p.drawRect(flagRect);

    // Sol de Mayo estilizado en el centro de la franja blanca
    const QPointF solCenter(midStripe.center());
    const float solR = qMin(midStripe.width(), midStripe.height()) * 0.22f;
    p.setBrush(QColor(255, 204, 0)); // amarillo
    p.setPen(Qt::NoPen);
    p.drawEllipse(solCenter, solR, solR);

    // Rayos del sol (líneas finas)
    p.setPen(QPen(QColor(255, 204, 0), 1.2f));
    const int rayos = 16;
    for (int i = 0; i < rayos; ++i) {
        const double ang = (i * (360.0 / rayos)) * M_PI / 180.0;
        const float r1 = solR * 1.05f;
        const float r2 = solR * 1.9f;
        QPointF a(solCenter.x() + r1 * cos(ang), solCenter.y() + r1 * sin(ang));
        QPointF b(solCenter.x() + r2 * cos(ang), solCenter.y() + r2 * sin(ang));
        p.drawLine(a, b);
    }
    // Pequeño círculo interior para dar detalle
    p.setBrush(QColor(255, 230, 120));
    p.setPen(Qt::NoPen);
    p.drawEllipse(solCenter, solR * 0.45f, solR * 0.45f);

    // Texto: "BASE MILITAR" (primera línea) centrado en la banda superior
    QRectF texto1Rect(banda.left() + emblemaSize + 8.0f, banda.top(), banda.width() - emblemaSize - 16.0f - flagW * 0.6f, banda.height());
    p.setFont(QFont("Sans", 14, QFont::Bold));
    // Sombra del texto
    p.setPen(QColor(0, 0, 0, 140));
    p.drawText(texto1Rect.translated(1.5f, 1.5f), Qt::AlignVCenter | Qt::AlignLeft, QString("BASE MILITAR"));
    // Texto principal
    p.setPen(QColor(240, 240, 230));
    p.drawText(texto1Rect, Qt::AlignVCenter | Qt::AlignLeft, QString("BASE MILITAR"));

    // Texto secundario: "CAMPO DE MAYO" (segunda línea) centrado en la parte inferior
    QRectF texto2Rect(cartelX + 6.0f, cartelY + cartelAlto * 0.48f, cartelAncho - 12.0f, cartelAlto * 0.44f);
    p.setFont(QFont("Sans", 11, QFont::DemiBold));
    // Sombra
    p.setPen(QColor(0, 0, 0, 120));
    p.drawText(texto2Rect.translated(1.0f, 1.0f), Qt::AlignCenter, QString("CAMPO DE MAYO"));
    // Texto principal
    p.setPen(QColor(230, 230, 220));
    p.drawText(texto2Rect, Qt::AlignCenter, QString("CAMPO DE MAYO"));

    // Pequeños detalles: líneas horizontales decorativas a ambos lados del texto secundario
    p.setPen(QPen(QColor(200, 200, 180, 120), 1));
    const float decoLen = 40.0f;
    const float midY = texto2Rect.center().y();
    p.drawLine(QPointF(texto2Rect.left() + 8.0f, midY), QPointF(texto2Rect.left() + 8.0f + decoLen, midY));
    p.drawLine(QPointF(texto2Rect.right() - 8.0f, midY), QPointF(texto2Rect.right() - 8.0f - decoLen, midY));

    p.restore();
}
float BaseMilitar::get_x() { return x_escenario; }
float BaseMilitar::get_y() { return y_escenario; }
float BaseMilitar::get_ancho() { return ancho_escenario; }

QPointF BaseMilitar::getPuntoSpawnInterno() const
{
    // Centro de la base
    return QPointF(x_escenario + ancho_escenario / 2.0f,
                   y_escenario + alto_escenario / 2.0f);
}

QPointF BaseMilitar::getPuntoSpawnExterno() const
{
    // Fuera de la base, al sur
    return QPointF(x_escenario + ancho_escenario / 2.0f,
                   y_escenario + alto_escenario + 200);
}
#ifndef CASCARUDO_H
#define CASCARUDO_H

#include "enemigos.h"
#include <QPainter>
#include <QPointF>

class Cascarudo : public Enemigo
{
public:
    Cascarudo(float x, float y);
    ~Cascarudo() = default;

    void dibujar(QPainter &painter) override;
    void actualizar() override;
    void moverHacia(QPointF objetivo);

private:
    float frameAnimacion;
    float velocidadAnimacion;
    bool animacionSubiendo;
    float anguloMovimiento;

    void dibujarArriba(QPainter &painter, float offset);
    void dibujarAbajo(QPainter &painter, float offset);
    void dibujarIzquierda(QPainter &painter, float offset);
    void dibujarDerecha(QPainter &painter, float offset);
    void dibujarDiagonal(QPainter &painter, float offset, int direccion);
};

#endif // CASCARUDO_H
#include "cascarudo.h"
#include <cmath>

Cascarudo::Cascarudo(float x, float y)
    : Enemigo(x, y), frameAnimacion(0.0f),
    velocidadAnimacion(0.08f), animacionSubiendo(true),
    anguloMovimiento(90.0f)
{
    ancho = 40.0f;
    alto = 40.0f;
    velocidad = 1.5f;
    sistemaVida = Vida(50.0f);  // Cascarudo con 50 HP
    danioContacto = 5.0f;       // Hace 5 de daño al contacto
}

void Cascarudo::dibujar(QPainter &painter)
{
    int pixelSize = 2;
    float offset = frameAnimacion * (pixelSize / 2.0f);

    // Normalizar ángulo entre 0 y 360
    float angulo = anguloMovimiento;
    while (angulo < 0) angulo += 360;
    while (angulo >= 360) angulo -= 360;

    // Determinar dirección basada en el ángulo (8 direcciones)
    if (angulo >= 337.5 || angulo < 22.5) {
        // Derecha (0°)
        dibujarDerecha(painter, offset);
    } else if (angulo >= 22.5 && angulo < 67.5) {
        // Abajo-Derecha (45°)
        dibujarDiagonal(painter, offset, 1);
    } else if (angulo >= 67.5 && angulo < 112.5) {
        // Abajo (90°)
        dibujarAbajo(painter, offset);
    } else if (angulo >= 112.5 && angulo < 157.5) {
        // Abajo-Izquierda (135°)
        dibujarDiagonal(painter, offset, 2);
    } else if (angulo >= 157.5 && angulo < 202.5) {
        // Izquierda (180°)
        dibujarIzquierda(painter, offset);
    } else if (angulo >= 202.5 && angulo < 247.5) {
        // Arriba-Izquierda (225°)
        dibujarDiagonal(painter, offset, 3);
    } else if (angulo >= 247.5 && angulo < 292.5) {
        // Arriba (270°)
        dibujarArriba(painter, offset);
    } else {
        // Arriba-Derecha (315°)
        dibujarDiagonal(painter, offset, 4);
    }
}

void Cascarudo::dibujarArriba(QPainter &painter, float offset)
{
    int pixelSize = 2;
    painter.setPen(Qt::NoPen);

    // Colores grises
    QColor caparazonOscuro(60, 60, 60);      // Gris oscuro
    QColor caparazonClaro(90, 90, 90);       // Gris medio
    QColor brillo(120, 120, 120);            // Gris claro para brillos
    QColor patas(30, 30, 30);                // Negro para patas
    QColor ojos(150, 0, 0);                  // Rojo para ojos

    // PATAS (3 pares)
    painter.setBrush(patas);
    // Par 1 (atrás)
    painter.drawRect(pos_x + 2*pixelSize, pos_y + 15*pixelSize + offset, 2*pixelSize, 8*pixelSize);
    painter.drawRect(pos_x + 16*pixelSize, pos_y + 15*pixelSize - offset, 2*pixelSize, 8*pixelSize);

    // Par 2 (medio)
    painter.drawRect(pos_x + 3*pixelSize, pos_y + 10*pixelSize - offset, 2*pixelSize, 6*pixelSize);
    painter.drawRect(pos_x + 15*pixelSize, pos_y + 10*pixelSize + offset, 2*pixelSize, 6*pixelSize);

    // Par 3 (delante)
    painter.drawRect(pos_x + 4*pixelSize, pos_y + 6*pixelSize + offset, 2*pixelSize, 5*pixelSize);
    painter.drawRect(pos_x + 14*pixelSize, pos_y + 6*pixelSize - offset, 2*pixelSize, 5*pixelSize);

    // CUERPO/CAPARAZÓN BASE
    painter.setBrush(caparazonOscuro);
    painter.drawRect(pos_x + 5*pixelSize, pos_y + 8*pixelSize, 10*pixelSize, 14*pixelSize);

    // CAPARAZÓN CON DETALLES (segmentos)
    painter.setBrush(caparazonClaro);
    painter.drawRect(pos_x + 6*pixelSize, pos_y + 9*pixelSize, 8*pixelSize, 4*pixelSize);
    painter.drawRect(pos_x + 6*pixelSize, pos_y + 14*pixelSize, 8*pixelSize, 4*pixelSize);
    painter.drawRect(pos_x + 6*pixelSize, pos_y + 19*pixelSize, 8*pixelSize, 3*pixelSize);

    // BRILLOS EN CAPARAZÓN
    painter.setBrush(brillo);
    painter.drawRect(pos_x + 7*pixelSize, pos_y + 10*pixelSize, 2*pixelSize, 2*pixelSize);
    painter.drawRect(pos_x + 11*pixelSize, pos_y + 15*pixelSize, 2*pixelSize, 2*pixelSize);

    // CABEZA
    painter.setBrush(caparazonOscuro);
    painter.drawRect(pos_x + 7*pixelSize, pos_y + 4*pixelSize, 6*pixelSize, 5*pixelSize);

    // OJOS ROJOS BRILLANTES
    painter.setBrush(ojos);
    painter.drawRect(pos_x + 8*pixelSize, pos_y + 5*pixelSize, 2*pixelSize, 2*pixelSize);
    painter.drawRect(pos_x + 11*pixelSize, pos_y + 5*pixelSize, 2*pixelSize, 2*pixelSize);

    // ANTENAS
    painter.setBrush(patas);
    painter.drawRect(pos_x + 7*pixelSize, pos_y + 2*pixelSize, 1*pixelSize, 3*pixelSize);
    painter.drawRect(pos_x + 12*pixelSize, pos_y + 2*pixelSize, 1*pixelSize, 3*pixelSize);

    // MANDÍBULAS
    painter.drawRect(pos_x + 8*pixelSize, pos_y + 8*pixelSize, 2*pixelSize, 1*pixelSize);
    painter.drawRect(pos_x + 11*pixelSize, pos_y + 8*pixelSize, 2*pixelSize, 1*pixelSize);
}

void Cascarudo::dibujarAbajo(QPainter &painter, float offset)
{
    int pixelSize = 2;
    painter.setPen(Qt::NoPen);

    QColor caparazonOscuro(60, 60, 60);
    QColor caparazonClaro(90, 90, 90);
    QColor brillo(120, 120, 120);
    QColor patas(30, 30, 30);

    // PATAS (3 pares) - vista desde atrás
    painter.setBrush(patas);
    // Par 1 (delante, ahora atrás en la vista)
    painter.drawRect(pos_x + 2*pixelSize, pos_y + 2*pixelSize - offset, 2*pixelSize, 8*pixelSize);
    painter.drawRect(pos_x + 16*pixelSize, pos_y + 2*pixelSize + offset, 2*pixelSize, 8*pixelSize);

    // Par 2 (medio)
    painter.drawRect(pos_x + 3*pixelSize, pos_y + 8*pixelSize + offset, 2*pixelSize, 6*pixelSize);
    painter.drawRect(pos_x + 15*pixelSize, pos_y + 8*pixelSize - offset, 2*pixelSize, 6*pixelSize);

    // Par 3 (atrás)
    painter.drawRect(pos_x + 4*pixelSize, pos_y + 13*pixelSize - offset, 2*pixelSize, 5*pixelSize);
    painter.drawRect(pos_x + 14*pixelSize, pos_y + 13*pixelSize + offset, 2*pixelSize, 5*pixelSize);

    // CUERPO/CAPARAZÓN (vista posterior)
    painter.setBrush(caparazonOscuro);
    painter.drawRect(pos_x + 5*pixelSize, pos_y + 4*pixelSize, 10*pixelSize, 14*pixelSize);

    // SEGMENTOS DEL CAPARAZÓN (desde atrás)
    painter.setBrush(caparazonClaro);
    painter.drawRect(pos_x + 6*pixelSize, pos_y + 5*pixelSize, 8*pixelSize, 3*pixelSize);
    painter.drawRect(pos_x + 6*pixelSize, pos_y + 9*pixelSize, 8*pixelSize, 4*pixelSize);
    painter.drawRect(pos_x + 6*pixelSize, pos_y + 14*pixelSize, 8*pixelSize, 4*pixelSize);

    // BRILLOS
    painter.setBrush(brillo);
    painter.drawRect(pos_x + 7*pixelSize, pos_y + 10*pixelSize, 2*pixelSize, 2*pixelSize);
    painter.drawRect(pos_x + 11*pixelSize, pos_y + 6*pixelSize, 2*pixelSize, 2*pixelSize);

    // PARTE TRASERA (abdomen)
    painter.setBrush(caparazonOscuro);

void Cascarudo::dibujarDiagonal(QPainter &painter, float offset, int direccion)
{
    int pixelSize = 2;
    painter.setPen(Qt::NoPen);

    QColor caparazonOscuro(60, 60, 60);
    QColor caparazonClaro(90, 90, 90);
    QColor brillo(120, 120, 120);
    QColor patas(30, 30, 30);
    QColor ojos(150, 0, 0);

    bool mirandoDerecha = (direccion == 1 || direccion == 4);
    bool mirandoArriba = (direccion == 3 || direccion == 4);

    // PATAS (vista diagonal - 3 patas visibles)
    painter.setBrush(patas);
    if (mirandoDerecha) {
        painter.drawRect(pos_x + 14*pixelSize, pos_y + 7*pixelSize - offset, 4*pixelSize, 2*pixelSize);
        painter.drawRect(pos_x + 14*pixelSize, pos_y + 13*pixelSize + offset, 4*pixelSize, 2*pixelSize);
        painter.drawRect(pos_x + 14*pixelSize, pos_y + 19*pixelSize - offset, 4*pixelSize, 2*pixelSize);
    } else {
        painter.drawRect(pos_x + 2*pixelSize, pos_y + 7*pixelSize + offset, 4*pixelSize, 2*pixelSize);
        painter.drawRect(pos_x + 2*pixelSize, pos_y + 13*pixelSize - offset, 4*pixelSize, 2*pixelSize);
        painter.drawRect(pos_x + 2*pixelSize, pos_y + 19*pixelSize + offset, 4*pixelSize, 2*pixelSize);
    }

    // CUERPO (oval diagonal)
    painter.setBrush(caparazonOscuro);
    if (mirandoArriba) {
        painter.drawRect(pos_x + 6*pixelSize, pos_y + 6*pixelSize, 8*pixelSize, 14*pixelSize);
    } else {
        painter.drawRect(pos_x + 6*pixelSize, pos_y + 6*pixelSize, 8*pixelSize, 14*pixelSize);
    }

    // SEGMENTOS
    painter.setBrush(caparazonClaro);
    painter.drawRect(pos_x + 7*pixelSize, pos_y + 8*pixelSize, 6*pixelSize, 4*pixelSize);
    painter.drawRect(pos_x + 7*pixelSize, pos_y + 13*pixelSize, 6*pixelSize, 3*pixelSize);
    painter.drawRect(pos_x + 7*pixelSize, pos_y + 17*pixelSize, 6*pixelSize, 2*pixelSize);

    // BRILLO
    painter.setBrush(brillo);
    painter.drawRect(pos_x + 8*pixelSize, pos_y + 9*pixelSize, 2*pixelSize, 2*pixelSize);

    // CABEZA
    painter.setBrush(caparazonOscuro);
    if (mirandoArriba) {
        painter.drawRect(pos_x + 7*pixelSize, pos_y + 3*pixelSize, 6*pixelSize, 4*pixelSize);
    } else {
        painter.drawRect(pos_x + 7*pixelSize, pos_y + 19*pixelSize, 6*pixelSize, 4*pixelSize);
    }

    // OJOS
    painter.setBrush(ojos);
    if (mirandoArriba) {
        if (mirandoDerecha) {
            painter.drawRect(pos_x + 10*pixelSize, pos_y + 4*pixelSize, 2*pixelSize, 2*pixelSize);
        } else {
            painter.drawRect(pos_x + 8*pixelSize, pos_y + 4*pixelSize, 2*pixelSize, 2*pixelSize);
        }
    } else {
        if (mirandoDerecha) {
            painter.drawRect(pos_x + 10*pixelSize, pos_y + 20*pixelSize, 2*pixelSize, 2*pixelSize);
        } else {
            painter.drawRect(pos_x + 8*pixelSize, pos_y + 20*pixelSize, 2*pixelSize, 2*pixelSize);
        }
    }
}

void Cascarudo::actualizar()
{
    // Animación de movimiento (patas)
    if (animacionSubiendo) {
        frameAnimacion += velocidadAnimacion;
    } else {
        frameAnimacion -= velocidadAnimacion;
    }

    if (frameAnimacion > 2.0f) {
        animacionSubiendo = false;
    }
    if (frameAnimacion < -2.0f) {
        animacionSubiendo = true;
    }
}

void Cascarudo::moverHacia(QPointF objetivo)
{
    // Calcular dirección hacia el objetivo
    float dx = objetivo.x() - (pos_x + ancho / 2.0f);
    float dy = objetivo.y() - (pos_y + alto / 2.0f);

    float distancia = std::sqrt(dx * dx + dy * dy);

    if (distancia > 5.0f) {
        // Calcular ángulo de movimiento
        anguloMovimiento = std::atan2(dy, dx) * 180.0f / M_PI;

        // Normalizar y aplicar velocidad
        dx = (dx / distancia) * velocidad;
        dy = (dy / distancia) * velocidad;

        mover(dx, dy);
    }
}

public:
    static bool hayColision(Dibujable_abstracto* obj1, Dibujable_abstracto* obj2);
    static bool hayColision(float x1, float y1, float ancho1, float alto1,
                            float x2, float y2, float ancho2, float alto2);
    static bool hayColisionCircular(float x1, float y1, float radio1,
                                    float x2, float y2, float radio2);
    static void resolverColision(Dibujable_abstracto* obj1, Dibujable_abstracto* obj2);

#include <algorithm>

bool Colision::hayColision(Dibujable_abstracto* obj1, Dibujable_abstracto* obj2)
{
    if (!obj1 || !obj2) return false;

    return hayColision(obj1->get_x(), obj1->get_y(), obj1->get_ancho(), obj1->get_alto(),
                       obj2->get_x(), obj2->get_y(), obj2->get_ancho(), obj2->get_alto());
}

bool Colision::hayColision(float x1, float y1, float ancho1, float alto1,
                           float x2, float y2, float ancho2, float alto2)
{
    return (x1 < x2 + ancho2 &&
            x1 + ancho1 > x2 &&
            y1 < y2 + alto2 &&
            y1 + alto1 > y2);
}

bool Colision::hayColisionCircular(float x1, float y1, float radio1,
                                   float x2, float y2, float radio2)
{
    float distancia = calcularDistancia(x1, y1, x2, y2);
    return distancia < (radio1 + radio2);
}

void Colision::resolverColision(Dibujable_abstracto* obj1, Dibujable_abstracto* obj2)
{
    if (!obj1 || !obj2) return;

    float x1 = obj1->get_x();
    float y1 = obj1->get_y();
    float ancho1 = obj1->get_ancho();
    float alto1 = obj1->get_alto();

    float x2 = obj2->get_x();
    float y2 = obj2->get_y();
    float ancho2 = obj2->get_ancho();
    float alto2 = obj2->get_alto();

    float centro1_x = x1 + ancho1 / 2.0f;
    float centro1_y = y1 + alto1 / 2.0f;
    float centro2_x = x2 + ancho2 / 2.0f;
    float centro2_y = y2 + alto2 / 2.0f;

    float dx = centro1_x - centro2_x;
    float dy = centro1_y - centro2_y;

    float overlapX = (ancho1 + ancho2) / 2.0f - std::abs(dx);
    float overlapY = (alto1 + alto2) / 2.0f - std::abs(dy);

    if (overlapX < overlapY) {
        if (dx > 0) {
            x1 += overlapX;
        } else {
            x1 -= overlapX;
        }
    } else {
        if (dy > 0) {
            y1 += overlapY;
        } else {
            y1 -= overlapY;
        }
    }
}

float Colision::calcularDistancia(float x1, float y1, float x2, float y2)
{
    float dx = x2 - x1;
    float dy = y2 - y1;
    return std::sqrt(dx * dx + dy * dy);
}

    float get_y() override { return pos_y; }
    float get_ancho() override { return radio * 2; }

    // Para saber cuándo eliminar el dibujo
    bool terminada() const { return tiempoVida <= 0; }

    float pos_x;
    float pos_y;
    float radio;
    float tiempoVida; // Contador regresivo
    float tiempoMax;  // Duración total (1.5s)

#include <cmath>

Explosivo::Explosivo(float x, float y, float radio)
    : pos_x(x), pos_y(y), radio(radio), tiempoMax(1.0f), tiempoVida(1.0f)
{
    // tiempoMax 1.0f es suficiente para un efecto rápido y visible
}

void Explosivo::actualizar()
{
    tiempoVida -= 0.016f; // Aprox 60 FPS
}

void Explosivo::dibujar(QPainter &painter)
{
    if (tiempoVida <= 0) return;

    painter.save();
    painter.setRenderHint(QPainter::Antialiasing, true);

    // Progreso de 0.0 (inicio) a 1.0 (fin)
    float progreso = 1.0f - (tiempoVida / tiempoMax);

    // El radio crece: empieza en 20% y termina en 100% del tamaño
    float radioAnimado = radio * (0.2f + 0.8f * progreso);

    // Calcular transparencia (Alpha)
    int alpha = (int)(255 * (1.0f - progreso));
    if (alpha < 0) alpha = 0;

    // --- DIBUJO ROJO BRILLANTE ---

    // 1. Círculo Exterior (Rojo Puro)
    QColor colorExterno(255, 0, 0, alpha); // ROJO
    painter.setBrush(colorExterno);
    painter.setPen(Qt::NoPen);
    painter.drawEllipse(QPointF(pos_x, pos_y), radioAnimado, radioAnimado);

    // 2. Círculo Interno (Amarillo/Naranja) - Para dar efecto de núcleo caliente
    float radioNucleo = radioAnimado * 0.7f;
    QColor colorNucleo(255, 200, 0, alpha); // AMARILLO
    painter.setBrush(colorNucleo);
    painter.drawEllipse(QPointF(pos_x, pos_y), radioNucleo, radioNucleo);

    painter.restore();
}

    painter.setBrush(QColor(45, 45, 50)); // Gris oscuro

    painter.drawRect(-15, -3, 5, 6);

    // Mira delantera
    painter.setBrush(QColor(200, 180, 50));
    painter.drawRect(28, -1, 2, 2);

    // Detalles metálicos
    painter.setBrush(QColor(90, 90, 100));
    painter.drawRect(-8, -4, 3, 2);

    painter.restore();

void Fal::actualizar(float posX, float posY, QPointF mouseWorld)
{
    pos_x = posX;
    pos_y = posY;

    float dx = mouseWorld.x() - pos_x;
    float dy = mouseWorld.y() - pos_y;
    angulo = std::atan2(dy, dx) * 180.0f / M_PI;

    if (contadorCadencia > 0) {
        contadorCadencia--;
    }

    actualizarRecarga();

    if (municionActual <= 0 && !recargando) {
        iniciarRecarga();
    }
}

void Fal::disparar()
{
    if (contadorCadencia == 0 && municionActual > 0 && !recargando) {
        // Proyectil FAL: velocidad 15.0, daño 15.0, NO explosivo
        Proyectil* p = new Proyectil(pos_x, pos_y, angulo, 15.0f, 15.0f);
        proyectiles.append(p);
        contadorCadencia = cadenciaDisparo;
        municionActual--;
    }
}

void Fal::actualizarProyectiles()
{
    for (int i = proyectiles.size() - 1; i >= 0; i--) {
        proyectiles[i]->actualizar();
        if (!proyectiles[i]->estaActivo()) {
            delete proyectiles[i];
            proyectiles.removeAt(i);
        }
    }
}

{
    return proyectiles;
}

void Gamelogic::actualizar() {

    if (!sobreviviente.estaVivo()) return;
    sobreviviente.actualizar(mouseWorldPos);

    if (estadoActual == EN_DESCANSO) {
        timerDescanso -= 0.016f;
        if (timerDescanso <= 0) iniciarNivel(nivelActual + 1);
    } else {
        bool hordaViva = false;
        for(auto z : zombies) if(z->estaVivo()) hordaViva = true;
        for(auto m : murcielagos) if(m->estaVivo()) hordaViva = true;
        for(auto c : cascarudos) if(c->estaVivo()) hordaViva = true;
        for(auto g : goliats) if(g->estaVivo()) hordaViva = true;
        if (!hordaViva) iniciarDescanso();
    }

    QPointF pJugador(sobreviviente.get_x(), sobreviviente.get_y());
    for(auto c : cascarudos) if(c->estaVivo()) { c->actualizar(); c->moverHacia(pJugador); }
    for(auto g : goliats) if(g->estaVivo()) { g->actualizar(); g->moverHacia(pJugador); }
    for(auto z : zombies) if(z->estaVivo()) { z->actualizar(); z->moverHacia(pJugador); }
    for(auto m : murcielagos) if(m->estaVivo()) { m->actualizar(); m->moverHacia(pJugador); }
    verificarColisiones();
    actualizarExplosiones();  // ← AGREGAR ESTA LÍNEA
    tiempoUltimoDanioContacto += 0.016f;
}




Tienda* Gamelogic::getTiendaCercana()
{
    float centroX = sobreviviente.get_x() + sobreviviente.get_ancho() / 2.0f;
    float centroY = sobreviviente.get_y() + sobreviviente.get_alto() / 2.0f;

    for(Tienda* t : tiendas) {
        if(t->estaEnRango(centroX, centroY)) {
            return t;
        }
    }
    return nullptr;
}

void Gamelogic::comprarItem(int itemId)
{
    Tienda* tienda = getTiendaCercana();
    if(!tienda || estadoActual != EN_DESCANSO) return;

    if(tienda->getTipo() == TIENDA_ARMAS) {
        // Items de Ferreterías Pipo
        int precio = 10;
        if(puntos < precio) return;

        switch(itemId) {
        case 1: // FAL
            if(tieneFal) return;
            puntos -= precio;
            tieneFal = true;
            sobreviviente.cambiarArma(2);
            break;
        case 2: // Granadas
            if(tieneGranadas) return;
            puntos -= precio;
            tieneGranadas = true;
            sobreviviente.cambiarArma(3);
            break;
        case 3: // Lanzacohetes
            if(tieneLanzacohetes) return;
            puntos -= precio;
            tieneLanzacohetes = true;
            sobreviviente.cambiarArma(4);
            break;
        }
    } else {
        // Items de Clínica Maldonado
        switch(itemId) {
        case 1: { // Agrandamiento de miembro (+20 HP máxima)
            int precio = 100;
            if(puntos < precio) return;
            puntos -= precio;
            nivelVidaExtra++;

            float nuevaVidaMax = sobreviviente.getVida().getVidaMaxima() + 20.0f;
            sobreviviente.getVida().setVidaMaxima(nuevaVidaMax);
            sobreviviente.getVida().curar(20.0f);
            break;
        }
        case 2: { // Implantes de senos (+10% velocidad)
            int precio = 150;
            if(puntos < precio) return;
            if(nivelVelocidadExtra >= 3) return; // Máximo 3 compras
            puntos -= precio;
            nivelVelocidadExtra++;
            break;
        }
        case 3: { // Vida (+50 HP)
            int precio = 50;
            if(puntos < precio) return;
            if(sobreviviente.getVida().getVidaActual() >= sobreviviente.getVida().getVidaMaxima()) {
                return;
            }
            puntos -= precio;
            sobreviviente.getVida().curar(50.0f);
            break;
        }
        }
    }
}

bool Gamelogic::tieneArma(int tipo) const
{
    switch(tipo) {
    case 2: return tieneFal;
    case 3: return tieneGranadas;
    case 4: return tieneLanzacohetes;
    default: return false;
    }
}

QString Gamelogic::getNombreArmaActual() const
{
    if(!sobreviviente.getArma()) return "NINGUNA";

    Arma* arma = sobreviviente.getArma();
    if(dynamic_cast<Pistola*>(arma)) return "PISTOLA";
    if(dynamic_cast<Fal*>(arma)) return "FAL";
    if(dynamic_cast<Granada*>(arma)) return "GRANADAS";
    if(dynamic_cast<Lanzacohetes*>(arma)) return "LANZACOHETES";

    return "DESCONOCIDA";
}

void Gamelogic::spawnEnemigos()
{
    // 1. Limpieza de enemigos anteriores de todas las listas
    // Limpiar Cascarudos
    for(auto c : cascarudos) Dibujables.removeOne(c);
    qDeleteAll(cascarudos);
    cascarudos.clear();

    // Limpiar Goliats
    for(auto g : goliats) Dibujables.removeOne(g);
    qDeleteAll(goliats);
    goliats.clear();

    // Limpiar Zombies con Pala
    for(auto z : zombies) Dibujables.removeOne(z);
    qDeleteAll(zombies);
    zombies.clear();

    // Limpiar Murciélagos
    for(auto m : murcielagos) Dibujables.removeOne(m);
    qDeleteAll(murcielagos);
    murcielagos.clear();

    // 2. Configuración de spawn aleatorio
    // Definimos el área de juego (ajusta según el tamaño de tu mapa)
    int margen = 100;
    int spawnWidth = 1200;
    int spawnHeight = 800;

    // --- SPAWN DE CASCARUDOS (Enemigo básico, siempre presente) ---
    int cantidadCascarudos = 5 + (nivelActual * 3);
    for (int i = 0; i < cantidadCascarudos; ++i) {
        float x = rand() % spawnWidth + margen;
        float y = rand() % spawnHeight + margen;
        Cascarudo* nuevo = new Cascarudo(x, y);
        cascarudos.append(nuevo);
        Dibujables.append(nuevo);
    }

    // --- SPAWN DE GOLIATS (Tanques, aparecen desde nivel 2) ---
    if (nivelActual >= 2) {
        int cantidadGoliats = nivelActual - 1;
        for (int i = 0; i < cantidadGoliats; ++i) {
            float x = rand() % spawnWidth + margen;
            float y = rand() % spawnHeight + margen;
            Goliat* nuevo = new Goliat(x, y);
            goliats.append(nuevo);
            Dibujables.append(nuevo);
        }
    }

    // --- SPAWN DE ZOMBIES CON PALA (Nuevos, desde nivel 2) ---
    if (nivelActual >= 2) {
        int cantidadZombies = 2 + (nivelActual * 2);
        for (int i = 0; i < cantidadZombies; ++i) {
            float x = rand() % spawnWidth + margen;
            float y = rand() % spawnHeight + margen;
            ZombiePala* nuevo = new ZombiePala(x, y);
            zombies.append(nuevo);
            Dibujables.append(nuevo);
        }
    }

    // --- SPAWN DE MURCIÉLAGOS (Rápidos, desde nivel 3) ---
    if (nivelActual >= 3) {
        int cantidadMurcielagos = 3 + (nivelActual * 3);
        for (int i = 0; i < cantidadMurcielagos; ++i) {
            float x = rand() % spawnWidth + margen;
            float y = rand() % spawnHeight + margen;
            Murcielago* nuevo = new Murcielago(x, y);
            murcielagos.append(nuevo);
            Dibujables.append(nuevo);
        }
    }

}

void Gamelogic::mover(int dx, int dy) {
    if (!sobreviviente.estaVivo()) return;
    float oldX = sobreviviente.get_x(), oldY = sobreviviente.get_y();

    // Aplicar multiplicador de velocidad + mejoras de velocidad
    float velocidadBase = 5.0f * sobreviviente.getVelocidadMovimiento();
    float velocidadTotal = velocidadBase * (1.0f + (nivelVelocidadExtra * 0.1f));
    sobreviviente.setPosicion(oldX + dx * velocidadTotal, oldY + dy * velocidadTotal);

    bool colision = false;

    for(auto p : base.obtenerParedes()) {
        if(Colision::hayColision(&sobreviviente, p)) {
            colision = true;
            break;
        }
    }

    if(colision) sobreviviente.setPosicion(oldX, oldY);
    sobreviviente.actualizarAnimacion(dx!=0||dy!=0, (dy<0)?1:0);
}

void Gamelogic::verificarColisionesProyectiles() {
    if (!sobreviviente.getArma()) return;

    QList<Proyectil*>& proyectiles = sobreviviente.getArma()->getProyectiles();

    for (int i = 0; i < proyectiles.size(); ++i) {
        Proyectil* p = proyectiles[i];
        if (!p->estaActivo()) continue;

        bool impacto = false;

        // --- COLISIÓN CON CASCARUDOS (Ya lo tienes) ---
        for (Cascarudo* c : cascarudos) {
            if (c->estaVivo() && Colision::hayColision(p, c)) {
                if(p->esExplosivo()) crearExplosion(p->get_x(), p->get_y(), p->getRadioExplosion(), p->getDanio());
                else { c->recibirDanio(p->getDanio()); if(!c->estaVivo()) puntos += 10; }
                p->desactivar(); impacto = true; break;
            }
        }
        if(impacto) continue;

        // --- NUEVO: COLISIÓN CON ZOMBIES CON PALA ---
        for (ZombiePala* z : zombies) {
            if (z->estaVivo() && Colision::hayColision(p, z)) {
                if(p->esExplosivo()) {
                    crearExplosion(p->get_x(), p->get_y(), p->getRadioExplosion(), p->getDanio());
                } else {
                    z->recibirDanio(p->getDanio());
                    if(!z->estaVivo()) puntos += 25; // Más puntos por ser más duro
                }
                p->desactivar(); impacto = true; break;
            }
        }
        if(impacto) continue;

        // --- NUEVO: COLISIÓN CON MURCIÉLAGOS ---
        for (Murcielago* m : murcielagos) {
            if (m->estaVivo() && Colision::hayColision(p, m)) {
                if(p->esExplosivo()) {
                    crearExplosion(p->get_x(), p->get_y(), p->getRadioExplosion(), p->getDanio());
                } else {
                    m->recibirDanio(p->getDanio());
                    if(!m->estaVivo()) puntos += 15;
                }
                p->desactivar(); impacto = true; break;
            }
        }
        if(impacto) continue;

        // --- COLISIÓN CON GOLIATS (Ya lo tienes) ---
        for (Goliat* g : goliats) {
            if (g->estaVivo() && Colision::hayColision(p, g)) {
                if(p->esExplosivo()) crearExplosion(p->get_x(), p->get_y(), p->getRadioExplosion(), p->getDanio());
                else { g->recibirDanio(p->getDanio()); if(!g->estaVivo()) puntos += 100; }
                p->desactivar(); impacto = true; break;
            }
        }
    }
}

void Gamelogic::verificarColisionesEnemigos() {
    // Colisiones entre enemigos para evitar solapamiento

    // Cascarudos entre sí
    for(int i = 0; i < cascarudos.size(); i++) {
        if(!cascarudos[i]->estaVivo()) continue;

        for(int j = i + 1; j < cascarudos.size(); j++) {
            if(!cascarudos[j]->estaVivo()) continue;

            if(Colision::hayColision(cascarudos[i], cascarudos[j])) {
                float dx = cascarudos[i]->get_x() - cascarudos[j]->get_x();
                float dy = cascarudos[i]->get_y() - cascarudos[j]->get_y();
                float dist = std::sqrt(dx * dx + dy * dy);
                if(dist > 0) {
                    cascarudos[i]->mover((dx / dist) * 1.5f, (dy / dist) * 1.5f);
                    cascarudos[j]->mover(-(dx / dist) * 1.5f, -(dy / dist) * 1.5f);
                }
            }
        }
    }

    // Goliats entre sí
    for(int i = 0; i < goliats.size(); i++) {
        if(!goliats[i]->estaVivo()) continue;

        for(int j = i + 1; j < goliats.size(); j++) {
            if(!goliats[j]->estaVivo()) continue;

            if(Colision::hayColision(goliats[i], goliats[j])) {
                float dx = goliats[i]->get_x() - goliats[j]->get_x();
                float dy = goliats[i]->get_y() - goliats[j]->get_y();
                float dist = std::sqrt(dx * dx + dy * dy);
                if(dist > 0) {
                    goliats[i]->mover((dx / dist) * 2.0f, (dy / dist) * 2.0f);
                    goliats[j]->mover(-(dx / dist) * 2.0f, -(dy / dist) * 2.0f);
                }
            }
        }
    }

    // Zombies entre sí
    for(int i = 0; i < zombies.size(); i++) {
        if(!zombies[i]->estaVivo()) continue;

        for(int j = i + 1; j < zombies.size(); j++) {
            if(!zombies[j]->estaVivo()) continue;

            if(Colision::hayColision(zombies[i], zombies[j])) {
                float dx = zombies[i]->get_x() - zombies[j]->get_x();
                float dy = zombies[i]->get_y() - zombies[j]->get_y();
                float dist = std::sqrt(dx * dx + dy * dy);
                if(dist > 0) {
                    zombies[i]->mover((dx / dist) * 1.5f, (dy / dist) * 1.5f);
                    zombies[j]->mover(-(dx / dist) * 1.5f, -(dy / dist) * 1.5f);
                }
            }
        }
    }

    // Murciélagos entre sí
    for(int i = 0; i < murcielagos.size(); i++) {
        if(!murcielagos[i]->estaVivo()) continue;

        for(int j = i + 1; j < murcielagos.size(); j++) {
            if(!murcielagos[j]->estaVivo()) continue;

            if(Colision::hayColision(murcielagos[i], murcielagos[j])) {
                float dx = murcielagos[i]->get_x() - murcielagos[j]->get_x();
                float dy = murcielagos[i]->get_y() - murcielagos[j]->get_y();
                float dist = std::sqrt(dx * dx + dy * dy);
                if(dist > 0) {
                    murcielagos[i]->mover((dx / dist) * 2.0f, (dy / dist) * 2.0f);
                    murcielagos[j]->mover(-(dx / dist) * 2.0f, -(dy / dist) * 2.0f);
                }
            }
        }
    }

    // Colisiones mixtas (entre diferentes tipos de enemigos)
    // Cascarudos con Goliats
    for(auto c : cascarudos) {
        if(!c->estaVivo()) continue;
        for(auto g : goliats) {
            if(!g->estaVivo()) continue;
            if(Colision::hayColision(c, g)) {
                float dx = c->get_x() - g->get_x();
                float dy = c->get_y() - g->get_y();
                float dist = std::sqrt(dx * dx + dy * dy);
                if(dist > 0) {
                    c->mover((dx / dist) * 2.0f, (dy / dist) * 2.0f);
                }
            }
        }
    }

    // Cascarudos con Zombies
    for(auto c : cascarudos) {
        if(!c->estaVivo()) continue;
        for(auto z : zombies) {
            if(!z->estaVivo()) continue;
            if(Colision::hayColision(c, z)) {
                float dx = c->get_x() - z->get_x();
                float dy = c->get_y() - z->get_y();
                float dist = std::sqrt(dx * dx + dy * dy);
                if(dist > 0) {
                    c->mover((dx / dist) * 1.5f, (dy / dist) * 1.5f);
                    z->mover(-(dx / dist) * 1.5f, -(dy / dist) * 1.5f);
                }
            }
        }
    }

    // Cascarudos con Murciélagos
    for(auto c : cascarudos) {
        if(!c->estaVivo()) continue;
        for(auto m : murcielagos) {
            if(!m->estaVivo()) continue;
            if(Colision::hayColision(c, m)) {
                float dx = c->get_x() - m->get_x();
                float dy = c->get_y() - m->get_y();
                float dist = std::sqrt(dx * dx + dy * dy);
                if(dist > 0) {
                    c->mover((dx / dist) * 1.5f, (dy / dist) * 1.5f);
                    m->mover(-(dx / dist) * 1.5f, -(dy / dist) * 1.5f);
                }
            }
        }
    }

    // Goliats con Zombies
    for(auto g : goliats) {
        if(!g->estaVivo()) continue;
        for(auto z : zombies) {
            if(!z->estaVivo()) continue;
            if(Colision::hayColision(g, z)) {
                float dx = z->get_x() - g->get_x();
                float dy = z->get_y() - g->get_y();
                float dist = std::sqrt(dx * dx + dy * dy);
                if(dist > 0) {
                    z->mover((dx / dist) * 2.0f, (dy / dist) * 2.0f);
                }
            }
        }
    }

    // Goliats con Murciélagos
    for(auto g : goliats) {
        if(!g->estaVivo()) continue;
        for(auto m : murcielagos) {
            if(!m->estaVivo()) continue;
            if(Colision::hayColision(g, m)) {
                float dx = m->get_x() - g->get_x();
                float dy = m->get_y() - g->get_y();
                float dist = std::sqrt(dx * dx + dy * dy);
                if(dist > 0) {
                    m->mover((dx / dist) * 2.0f, (dy / dist) * 2.0f);
                }
            }
        }
    }

    // Zombies con Murciélagos
    for(auto z : zombies) {
        if(!z->estaVivo()) continue;
        for(auto m : murcielagos) {
            if(!m->estaVivo()) continue;
            if(Colision::hayColision(z, m)) {
                float dx = z->get_x() - m->get_x();
                float dy = z->get_y() - m->get_y();
                float dist = std::sqrt(dx * dx + dy * dy);
                if(dist > 0) {
                    z->mover((dx / dist) * 1.5f, (dy / dist) * 1.5f);
                    m->mover(-(dx / dist) * 1.5f, -(dy / dist) * 1.5f);
                }
            }
        }
    }

    // Daño de Zombies con Pala al sobreviviente
    for (ZombiePala* z : zombies) {
        if (z->estaVivo() && Colision::hayColision(&sobreviviente, z)) {
            sobreviviente.recibirDanio(z->getDanioContacto());
        }
    }

    // Daño de Murciélagos al sobreviviente
    for (Murcielago* m : murcielagos) {
        if (m->estaVivo() && Colision::hayColision(&sobreviviente, m)) {
            sobreviviente.recibirDanio(m->getDanioContacto());
        }
        if(impacto) continue;

        // --- COLISIÓN CON MUTANTES ---
        for (Mutante* mut : mutantes) {
            if (mut->estaVivo() && Colision::hayColision(p, mut)) {
                if(p->esExplosivo()) {
                    crearExplosion(p->get_x(), p->get_y(), p->getRadioExplosion(), p->getDanio());
                } else {
                    mut->recibirDanio(p->getDanio());
                    if(!mut->estaVivo()) puntos += 150; // Muchos puntos por ser jefe
                }
                p->desactivar(); impacto = true; break;
            }
        }
    }
}
void Gamelogic::verificarColisiones() {
    verificarColisionesProyectiles();
    verificarColisionesEnemigos();
    verificarColisionConSobreviviente();
}

void Gamelogic::verificarColisionConSobreviviente() {
    if (tiempoUltimoDanioContacto < 0.5f) return;

    for(auto c : cascarudos) {
        if(c->estaVivo() && Colision::hayColision(&sobreviviente, c)) {
            sobreviviente.recibirDanio(c->getDanioContacto());
            tiempoUltimoDanioContacto = 0;
            return;
        }
    }

    for(auto g : goliats) {
        if(g->estaVivo() && Colision::hayColision(&sobreviviente, g)) {
            sobreviviente.recibirDanio(g->getDanioContacto());
            tiempoUltimoDanioContacto = 0;
            return;
        }
    }
}
void Gamelogic::crearExplosion(float x, float y, float radio, float danio)
{
    // 1. Crear efecto visual (Lo agregamos a Dibujables para que se vea)
    Explosivo* exp = new Explosivo(x, y, radio);
    explosiones.append(exp);
    Dibujables.append(exp); // IMPORTANTE: Se agrega al final para dibujarse ENCIMA del fondo

    // 2. Aplicar daño en área a CASCARUDOS
    for(auto c : cascarudos) {
        if(!c->estaVivo()) continue;

        // Calculamos la distancia desde el centro de la explosión (x,y)
        // hasta el CENTRO del cascarudo
        float enemigoCentroX = c->get_x() + c->get_ancho() / 2.0f;
        float enemigoCentroY = c->get_y() + c->get_alto() / 2.0f;

        float dx = enemigoCentroX - x;
        float dy = enemigoCentroY - y;

        // Distancia euclidiana
        float distancia = std::sqrt(dx * dx + dy * dy);

        // Si está dentro del radio, recibe daño
        if(distancia <= radio) {
            // Hacemos daño completo o reducido según prefieras.
            // Aquí simplificado para asegurar que mate:
            c->recibirDanio(danio);

            if(!c->estaVivo()) puntos += 10;
        }
    }

    // 3. Aplicar daño en área a GOLIATS
    for(auto g : goliats) {
        if(!g->estaVivo()) continue;

        float enemigoCentroX = g->get_x() + g->get_ancho() / 2.0f;
        float enemigoCentroY = g->get_y() + g->get_alto() / 2.0f;
        float dx = enemigoCentroX - x;
        float dy = enemigoCentroY - y;
        float distancia = std::sqrt(dx * dx + dy * dy);

        if(distancia <= radio) {
            g->recibirDanio(danio);
            if(!g->estaVivo()) puntos += 100;
        }
    }
    // 4. Aplicar daño en área a ZOMBIES CON PALA
    for(auto z : zombies) {
        if(!z->estaVivo()) continue;

        float enemigoCentroX = z->get_x() + z->get_ancho() / 2.0f;
        float enemigoCentroY = z->get_y() + z->get_alto() / 2.0f;
        float dx = enemigoCentroX - x;
        float dy = enemigoCentroY - y;
        float distancia = std::sqrt(dx * dx + dy * dy);

        if(distancia <= radio) {
            z->recibirDanio(danio);
            if(!z->estaVivo()) puntos += 25;
        }
    }

    // 5. Aplicar daño en área a MURCIÉLAGOS
    for(auto m : murcielagos) {
        if(!m->estaVivo()) continue;

        float enemigoCentroX = m->get_x() + m->get_ancho() / 2.0f;
        float enemigoCentroY = m->get_y() + m->get_alto() / 2.0f;
        float dx = enemigoCentroX - x;
        float dy = enemigoCentroY - y;
        float distancia = std::sqrt(dx * dx + dy * dy);

        if(distancia <= radio) {
            m->recibirDanio(danio);
            if(!m->estaVivo()) puntos += 15;
        }
    }
}

void Gamelogic::actualizarExplosiones()
{
    for(int i = explosiones.size() - 1; i >= 0; i--) {
        explosiones[i]->actualizar();
        if(explosiones[i]->terminada()) {
            Dibujables.removeOne(explosiones[i]);
            delete explosiones[i];
            explosiones.removeAt(i);
        }
    }
}

Gamewidget::Gamewidget(Gamelogic &gl, QWidget *parent)
    : QWidget(parent), gamelogic(gl), enSeleccionPersonaje(true),
    skinSeleccionada(SKIN_ETERNAUTA), itemSeleccionadoTienda(0)
{
    Crear_ventana();

    timer_principal = new QTimer(this);
    connect(timer_principal, &QTimer::timeout, this, [this]() {
        if (!enSeleccionPersonaje) {
            int dx = 0, dy = 0;
            if (teclasPresionadas.contains(Qt::Key_W)) dy -= 1;
            if (teclasPresionadas.contains(Qt::Key_S)) dy += 1;
            if (teclasPresionadas.contains(Qt::Key_A)) dx -= 1;
            if (teclasPresionadas.contains(Qt::Key_D)) dx += 1;

            this->gamelogic.setMouseWorldPos(this->mouseWorldPos);
            this->gamelogic.mover(dx, dy);
            this->gamelogic.actualizar();
        }
        this->update();
    });
    timer_principal->start(16);

    setFocusPolicy(Qt::StrongFocus);
    setMouseTracking(true);

    camPos = QPointF(0, 0);
    mouseScreenPos = QPointF(width() / 2.0, height() / 2.0);
}

}

void Gamewidget::dibujarItemTienda(QPainter &painter, int x, int y, QString nombre, QString desc,
                                   QString precio, bool seleccionado, bool yaComprado)
{
    int itemAncho = 560;
    int itemAlto = 60;

    QColor colorFondo = seleccionado ? QColor(80, 120, 80, 180) : QColor(50, 60, 70, 150);
    if(yaComprado) colorFondo = QColor(100, 100, 100, 120);

    painter.fillRect(x, y, itemAncho, itemAlto, colorFondo);

    if(seleccionado && !yaComprado) {
        painter.setPen(QPen(QColor(150, 255, 150), 3));
    } else {
        painter.setPen(QPen(QColor(80, 80, 90), 1));
    }
    painter.drawRect(x, y, itemAncho, itemAlto);

    if(yaComprado) {
        painter.setPen(QColor(150, 150, 150));
    } else {
        painter.setPen(QColor(255, 255, 255));
    }

    painter.setFont(QFont("Arial", 14, QFont::Bold));
    painter.drawText(x + 10, y + 25, nombre);

    painter.setFont(QFont("Arial", 10));
    painter.drawText(x + 10, y + 45, desc);

    if(yaComprado) {
        painter.setPen(QColor(100, 200, 100));
        painter.setFont(QFont("Arial", 12, QFont::Bold));
        painter.drawText(x + itemAncho - 120, y + 35, "✓ COMPRADO");
    } else {
        painter.setPen(QColor(255, 200, 100));
        painter.setFont(QFont("Arial", 14, QFont::Bold));
        painter.drawText(x + itemAncho - 100, y + 35, precio);
    }
}

void Gamewidget::paintEvent(QPaintEvent *event)
{
    QPainter painter(this);
    painter.setRenderHint(QPainter::Antialiasing);

    if (enSeleccionPersonaje) {
        dibujarSeleccionPersonaje(painter);
        return;
    }

    QList<Dibujable_abstracto*> &lista = gamelogic.Obtener_Dibujables();

    Sobreviviente* sobreviviente = nullptr;
    for (Dibujable_abstracto* d : lista) {
        if (auto* s = dynamic_cast<Sobreviviente*>(d)) {
            sobreviviente = s;
            break;
        }
    }

    }

    mouseWorldPos = camPos + mouseScreenPos;

    painter.save();
    painter.translate(-camPos.x(), -camPos.y());

    for (Dibujable_abstracto *d : lista) {
        if (d) d->dibujar(painter);
    }

    if (sobreviviente && sobreviviente->getArma()) {
        auto& proyectiles = sobreviviente->getArma()->getProyectiles();
        for (auto* p : proyectiles) {

        }

    // HUD Principal
    painter.fillRect(0, 0, width(), 70, QColor(20, 25, 35, 200));

                                      QString::number(arma->getMunicionMaxima()));

        if(arma->estaRecargando()) {

            int barraY = 55;

            int barraAlto = 8;

            painter.drawRect(barraX, barraY, barraAncho, barraAlto);

            painter.setBrush(QColor(255, 150, 0));
            painter.drawRect(barraX, barraY, barraAncho * progreso, barraAlto);

            painter.setPen(QPen(Qt::black, 1));
            painter.drawRect(barraX, barraY, barraAncho, barraAlto);

}

void Gamewidget::keyPressEvent(QKeyEvent *event)
{
    if (enSeleccionPersonaje) {
        if (event->key() == Qt::Key_Right) {
            if (skinSeleccionada == SKIN_ETERNAUTA) skinSeleccionada = SKIN_BRISA;
            else if (skinSeleccionada == SKIN_BRISA) skinSeleccionada = SKIN_MILITAR;
            update();
            return;
        }
        if (event->key() == Qt::Key_Left) {
            if (skinSeleccionada == SKIN_MILITAR) skinSeleccionada = SKIN_BRISA;
            else if (skinSeleccionada == SKIN_BRISA) skinSeleccionada = SKIN_ETERNAUTA;
            update();
            return;
        }
        if (event->key() == Qt::Key_Return || event->key() == Qt::Key_Enter) {
            enSeleccionPersonaje = false;
            // Aplicar skin al sobreviviente en la lógica del juego
            gamelogic.setSkinSobreviviente(skinSeleccionada);
            return;
        }
    }


    Tienda* tiendaCercana = gamelogic.getTiendaCercana();
    if(tiendaCercana && gamelogic.getEstado() == EN_DESCANSO) {
        if(event->key() == Qt::Key_Up) {
            itemSeleccionadoTienda--;
            if(itemSeleccionadoTienda < 0) itemSeleccionadoTienda = 2;
            return;
        } else if(event->key() == Qt::Key_Down) {
            itemSeleccionadoTienda++;
            if(itemSeleccionadoTienda > 2) itemSeleccionadoTienda = 0;
            return;
        } else if(event->key() == Qt::Key_E) {
            gamelogic.comprarItem(itemSeleccionadoTienda + 1);
            return;
        } else if(event->key() == Qt::Key_Escape) {
            return;
        }
    }

    if (event->key() == Qt::Key_1) {
        gamelogic.cambiarArma(1);
    } else if (event->key() == Qt::Key_2) {
        if(gamelogic.tieneArma(2)) gamelogic.cambiarArma(2);
    } else if (event->key() == Qt::Key_3) {
        if(gamelogic.tieneArma(3)) gamelogic.cambiarArma(3);
    } else if (event->key() == Qt::Key_4) {
        if(gamelogic.tieneArma(4)) gamelogic.cambiarArma(4);
    }

    // Agregar tecla al conjunto de teclas presionadas
    teclasPresionadas.insert(event->key());
}

void Gamewidget::keyReleaseEvent(QKeyEvent *event)
{
    // Quitar tecla del conjunto de teclas presionadas
    teclasPresionadas.remove(event->key());
}

void Gamewidget::mouseMoveEvent(QMouseEvent *event)
{
    mouseScreenPos = QPointF(event->pos().x(), event->pos().y());
}

void Gamewidget::mousePressEvent(QMouseEvent *event)
{
    if (event->button() == Qt::LeftButton && !enSeleccionPersonaje) {
        gamelogic.disparar();
    }
}
#ifndef GOLIAT_H
#define GOLIAT_H

#include "enemigos.h"
#include <QPainter>
#include <QPointF>

class Goliat : public Enemigo
{
public:
    Goliat(float x, float y);
    ~Goliat() = default;

    void dibujar(QPainter &painter) override;
    void actualizar() override;
    void moverHacia(QPointF objetivo);

private:
    float frameAnimacion;
    float velocidadAnimacion;
    bool animacionSubiendo;
    float anguloMovimiento;

    void dibujarArriba(QPainter &painter, float offset);
    void dibujarAbajo(QPainter &painter, float offset);
    void dibujarIzquierda(QPainter &painter, float offset);
    void dibujarDerecha(QPainter &painter, float offset);
    void dibujarDiagonal(QPainter &painter, float offset, int direccion);
};

#endif // GOLIAT_H
#include "goliat.h"
#include <cmath>

Goliat::Goliat(float x, float y)
    : Enemigo(x, y), frameAnimacion(0.0f),
    velocidadAnimacion(0.05f), animacionSubiendo(true),
    anguloMovimiento(90.0f)
{
    ancho = 60.0f;
    alto = 60.0f;
    velocidad = 0.8f;
    sistemaVida = Vida(150.0f);  // Goliat con 150 HP
    danioContacto = 15.0f;        // Hace 15 de daño al contacto
}

void Goliat::dibujar(QPainter &painter)
{
    int pixelSize = 3;
    float offset = frameAnimacion * (pixelSize / 2.0f);

    // Normalizar ángulo entre 0 y 360
    float angulo = anguloMovimiento;
    while (angulo < 0) angulo += 360;
    while (angulo >= 360) angulo -= 360;

    // Determinar dirección basada en el ángulo (8 direcciones)
    if (angulo >= 337.5 || angulo < 22.5) {
        dibujarDerecha(painter, offset);
    } else if (angulo >= 22.5 && angulo < 67.5) {
        dibujarDiagonal(painter, offset, 1);
    } else if (angulo >= 67.5 && angulo < 112.5) {
        dibujarAbajo(painter, offset);
    } else if (angulo >= 112.5 && angulo < 157.5) {
        dibujarDiagonal(painter, offset, 2);
    } else if (angulo >= 157.5 && angulo < 202.5) {
        dibujarIzquierda(painter, offset);
    } else if (angulo >= 202.5 && angulo < 247.5) {
        dibujarDiagonal(painter, offset, 3);
    } else if (angulo >= 247.5 && angulo < 292.5) {
        dibujarArriba(painter, offset);
    } else {
        dibujarDiagonal(painter, offset, 4);
    }
}

void Goliat::dibujarDiagonal(QPainter &painter, float offset, int direccion)
{
    int pixelSize = 3;
    painter.setPen(Qt::NoPen);

    QColor cuerpoOscuro(40, 40, 50);
    QColor cuerpoMedio(55, 55, 65);
    QColor armadura(70, 70, 80);
    QColor patas(25, 25, 25);
    QColor ojos(200, 50, 0);

    bool mirandoDerecha = (direccion == 1 || direccion == 4);
    bool mirandoArriba = (direccion == 3 || direccion == 4);

    // PATAS (vista diagonal - 4 patas visibles)
    painter.setBrush(patas);
    if (mirandoDerecha) {
        painter.drawRect(pos_x + 22*pixelSize, pos_y + 9*pixelSize - offset, 5*pixelSize, 3*pixelSize);
        painter.drawRect(pos_x + 22*pixelSize, pos_y + 17*pixelSize + offset, 5*pixelSize, 3*pixelSize);
        painter.drawRect(pos_x + 22*pixelSize, pos_y + 25*pixelSize - offset, 5*pixelSize, 3*pixelSize);
        painter.drawRect(pos_x + 22*pixelSize, pos_y + 33*pixelSize + offset, 5*pixelSize, 3*pixelSize);
    } else {
        painter.drawRect(pos_x + 3*pixelSize, pos_y + 9*pixelSize + offset, 5*pixelSize, 3*pixelSize);
        painter.drawRect(pos_x + 3*pixelSize, pos_y + 17*pixelSize - offset, 5*pixelSize, 3*pixelSize);
        painter.drawRect(pos_x + 3*pixelSize, pos_y + 25*pixelSize + offset, 5*pixelSize, 3*pixelSize);
        painter.drawRect(pos_x + 3*pixelSize, pos_y + 33*pixelSize - offset, 5*pixelSize, 3*pixelSize);
    }

    // CUERPO (diagonal)
    painter.setBrush(cuerpoOscuro);
    if (mirandoArriba) {
        painter.drawRect(pos_x + 9*pixelSize, pos_y + 8*pixelSize, 12*pixelSize, 20*pixelSize);
    } else {
        painter.drawRect(pos_x + 9*pixelSize, pos_y + 10*pixelSize, 12*pixelSize, 20*pixelSize);
    }

    // SEGMENTOS
    painter.setBrush(armadura);
    int offsetY = mirandoArriba ? 0 : 2;
    painter.drawRect(pos_x + 10*pixelSize, pos_y + (10 + offsetY)*pixelSize, 10*pixelSize, 6*pixelSize);
    painter.drawRect(pos_x + 10*pixelSize, pos_y + (17 + offsetY)*pixelSize, 10*pixelSize, 5*pixelSize);
    painter.drawRect(pos_x + 10*pixelSize, pos_y + (23 + offsetY)*pixelSize, 10*pixelSize, 4*pixelSize);

    // CABEZA
    painter.setBrush(cuerpoMedio);
    if (mirandoArriba) {
        painter.drawRect(pos_x + 10*pixelSize, pos_y + 4*pixelSize, 10*pixelSize, 5*pixelSize);
    } else {
        painter.drawRect(pos_x + 10*pixelSize, pos_y + 30*pixelSize, 10*pixelSize, 5*pixelSize);
    }

    // OJOS
    painter.setBrush(ojos);
    if (mirandoArriba) {
        if (mirandoDerecha) {
            painter.drawRect(pos_x + 15*pixelSize, pos_y + 5*pixelSize, 3*pixelSize, 3*pixelSize);
        } else {
            painter.drawRect(pos_x + 12*pixelSize, pos_y + 5*pixelSize, 3*pixelSize, 3*pixelSize);
        }
    } else {
        if (mirandoDerecha) {
            painter.drawRect(pos_x + 15*pixelSize, pos_y + 31*pixelSize, 3*pixelSize, 3*pixelSize);
        } else {
            painter.drawRect(pos_x + 12*pixelSize, pos_y + 31*pixelSize, 3*pixelSize, 3*pixelSize);
        }
    }

    // CUERNO
    painter.setBrush(patas);
    if (mirandoArriba) {
        painter.drawRect(pos_x + 13*pixelSize, pos_y + 2*pixelSize, 3*pixelSize, 3*pixelSize);
    }
}

void Goliat::actualizar()
{
    // Animación de movimiento (más lenta)
    if (animacionSubiendo) {
        frameAnimacion += velocidadAnimacion;
    } else {
        frameAnimacion -= velocidadAnimacion;
    }

    if (frameAnimacion > 1.5f) {
        animacionSubiendo = false;
    }
    if (frameAnimacion < -1.5f) {
        animacionSubiendo = true;
    }
}

void Goliat::moverHacia(QPointF objetivo)
{
    // Calcular dirección hacia el objetivo
    float dx = objetivo.x() - (pos_x + ancho / 2.0f);
    float dy = objetivo.y() - (pos_y + alto / 2.0f);

    float distancia = std::sqrt(dx * dx + dy * dy);

    if (distancia > 5.0f) {
        // Calcular ángulo de movimiento
        anguloMovimiento = std::atan2(dy, dx) * 180.0f / M_PI;

        // Normalizar y aplicar velocidad
        dx = (dx / distancia) * velocidad;
        dy = (dy / distancia) * velocidad;

        mover(dx, dy);
    }
}

void Granada::dibujar(QPainter &painter)

    painter.save();
    painter.translate(pos_x, pos_y);

    painter.setBrush(QColor(60, 80, 60));

void Granada::actualizar(float posX, float posY, QPointF mouseWorld)
{
    pos_x = posX;
    pos_y = posY;

    float dx = mouseWorld.x() - pos_x;
    float dy = mouseWorld.y() - pos_y;
    angulo = std::atan2(dy, dx) * 180.0f / M_PI;

    if (contadorCadencia > 0) {
        contadorCadencia--;
    }

    actualizarRecarga();

    if (municionActual <= 0 && !recargando) {
        iniciarRecarga();
    }
}

void Granada::disparar()
{
    if (contadorCadencia == 0 && municionActual > 0 && !recargando) {
        // CORRECCIÓN: Agregar 'true' (es explosivo) y '120.0f' (radio de explosión)
        // Argumentos: x, y, angulo, velocidad, daño, ES_EXPLOSIVO, RADIO
        Proyectil* p = new Proyectil(pos_x, pos_y, angulo, 6.0f, 60.0f, true, 120.0f);
        proyectiles.append(p);
        contadorCadencia = cadenciaDisparo;
        municionActual--;
    }
}

void Granada::actualizarProyectiles()
{
    for (int i = proyectiles.size() - 1; i >= 0; i--) {
        proyectiles[i]->actualizar();
        if (!proyectiles[i]->estaActivo()) {
            delete proyectiles[i];
            proyectiles.removeAt(i);
        }
    }
}

QList<Proyectil*>& Granada::getProyectiles()
{
    return proyectiles;
}

#ifndef MUTANTE_H

class Mutante : public Dibujable_abstracto
{
public:
    Mutante(float x, float y);
    ~Mutante() = default;

    void dibujar(QPainter &painter) override;
    void actualizar();
    void moverHacia(QPointF objetivo);
    void mover(float dx, float dy);
    void recibirDanio(float danio);

    float get_alto() override { return alto; }

private:
    float pos_x;
    float pos_y;
    float ancho;
    float alto;
    float velocidad;
    float danioContacto;

    Vida sistemaVida;

    // Sistema de animación
    float frameAnimacion;
    float velocidadAnimacion;
    bool animacionSubiendo;

    // Animación de tentáculos
    float anguloTentaculo1;
    float anguloTentaculo2;
    float anguloTentaculo3;
    float velocidadTentaculos;

#include "mutante.h"

Mutante::Mutante(float x, float y)
    : pos_x(x), pos_y(y), ancho(50.0f), alto(60.0f),
    velocidad(1.8f), danioContacto(40.0f), sistemaVida(300.0f),
    frameAnimacion(0.0f), velocidadAnimacion(0.08f), animacionSubiendo(true),
    anguloTentaculo1(0.0f), anguloTentaculo2(120.0f), anguloTentaculo3(240.0f),
    velocidadTentaculos(0.15f)
{
}

void Mutante::dibujar(QPainter &painter)
{
    if (!sistemaVida.estaVivo()) return;

    painter.save();
    painter.setPen(Qt::NoPen);

    int pixelSize = 2;
    float offset = frameAnimacion * (pixelSize / 2.0f);

    // Colores violetas
    QColor violetaOscuro(80, 40, 120);
    QColor violetaClaro(120, 60, 180);
    QColor violetaTentaculo(100, 50, 150);
    QColor negro(20, 10, 30);

    // PIERNAS (con animación de movimiento)
    painter.setBrush(violetaOscuro);
    // Pierna izquierda
    painter.drawRect(pos_x + 8*pixelSize, pos_y + 22*pixelSize + offset, 5*pixelSize, 12*pixelSize);
    // Pierna derecha
    painter.drawRect(pos_x + 15*pixelSize, pos_y + 22*pixelSize - offset, 5*pixelSize, 12*pixelSize);

    // Pies más oscuros
    painter.setBrush(negro);
    painter.drawRect(pos_x + 7*pixelSize, pos_y + 33*pixelSize + offset, 6*pixelSize, 3*pixelSize);
    painter.drawRect(pos_x + 14*pixelSize, pos_y + 33*pixelSize - offset, 6*pixelSize, 3*pixelSize);

    // CUERPO PRINCIPAL (grande y robusto)
    painter.setBrush(violetaClaro);
    painter.drawRect(pos_x + 6*pixelSize, pos_y + 10*pixelSize, 16*pixelSize, 14*pixelSize);

    // Detalles del cuerpo (manchas más oscuras)
    painter.setBrush(violetaOscuro);
    painter.drawRect(pos_x + 9*pixelSize, pos_y + 13*pixelSize, 4*pixelSize, 4*pixelSize);
    painter.drawRect(pos_x + 15*pixelSize, pos_y + 16*pixelSize, 4*pixelSize, 4*pixelSize);

    // TENTÁCULOS (3 tentáculos animados)
    painter.setBrush(violetaTentaculo);

    // Tentáculo 1 (izquierdo) - con animación ondulante
    float desvioT1 = sin(anguloTentaculo1 * M_PI / 180.0f) * 2;
    painter.drawRect(pos_x + 2*pixelSize + desvioT1, pos_y + 12*pixelSize, 4*pixelSize, 3*pixelSize);
    painter.drawRect(pos_x + 1*pixelSize + desvioT1, pos_y + 15*pixelSize, 3*pixelSize, 4*pixelSize);
    painter.drawRect(pos_x + 0*pixelSize + desvioT1*1.5f, pos_y + 19*pixelSize, 2*pixelSize, 3*pixelSize);

    // Tentáculo 2 (centro-superior) - con animación ondulante
    float desvioT2 = sin(anguloTentaculo2 * M_PI / 180.0f) * 2;
    painter.drawRect(pos_x + 11*pixelSize, pos_y + 6*pixelSize + desvioT2, 6*pixelSize, 4*pixelSize);
    painter.drawRect(pos_x + 12*pixelSize, pos_y + 3*pixelSize + desvioT2*1.5f, 4*pixelSize, 3*pixelSize);
    painter.drawRect(pos_x + 13*pixelSize, pos_y + 0*pixelSize + desvioT2*2, 2*pixelSize, 3*pixelSize);

    // Tentáculo 3 (derecho) - con animación ondulante
    float desvioT3 = sin(anguloTentaculo3 * M_PI / 180.0f) * 2;
    painter.drawRect(pos_x + 22*pixelSize - desvioT3, pos_y + 13*pixelSize, 4*pixelSize, 3*pixelSize);
    painter.drawRect(pos_x + 24*pixelSize - desvioT3, pos_y + 16*pixelSize, 3*pixelSize, 4*pixelSize);
    painter.drawRect(pos_x + 26*pixelSize - desvioT3*1.5f, pos_y + 20*pixelSize, 2*pixelSize, 3*pixelSize);

    // Puntas de tentáculos (más oscuras)
    painter.setBrush(violetaOscuro);
    painter.drawEllipse(QPointF(pos_x + 1*pixelSize + desvioT1*1.5f, pos_y + 21*pixelSize), 2, 2);
    painter.drawEllipse(QPointF(pos_x + 14*pixelSize, pos_y + 1*pixelSize + desvioT2*2), 2, 2);
    painter.drawEllipse(QPointF(pos_x + 27*pixelSize - desvioT3*1.5f, pos_y + 22*pixelSize), 2, 2);

    // CABEZA / OJO CENTRAL
    painter.setBrush(violetaOscuro);
    painter.drawRect(pos_x + 10*pixelSize, pos_y + 8*pixelSize, 8*pixelSize, 6*pixelSize);

    // Ojo brillante (rojo)
    painter.setBrush(QColor(200, 0, 0));
    painter.drawEllipse(QPointF(pos_x + 14*pixelSize, pos_y + 11*pixelSize), 3, 3);
    painter.setBrush(QColor(255, 50, 50));
    painter.drawEllipse(QPointF(pos_x + 14*pixelSize, pos_y + 11*pixelSize), 1.5, 1.5);

    painter.restore();

    // Barra de vida
    if (sistemaVida.estaVivo()) {
        float barraAncho = 50.0f;
        float barraAlto = 5.0f;
        float barraX = pos_x;
        float barraY = pos_y - 10.0f;

        painter.setBrush(QColor(100, 0, 0));
        painter.drawRect(barraX, barraY, barraAncho, barraAlto);

        float porcentaje = sistemaVida.getPorcentajeVida() / 100.0f;
        QColor colorVida = sistemaVida.estaCritico() ? QColor(200, 50, 0) : QColor(150, 0, 200);
        painter.setBrush(colorVida);
        painter.drawRect(barraX, barraY, barraAncho * porcentaje, barraAlto);

        painter.setBrush(Qt::NoBrush);
        painter.setPen(QPen(Qt::black, 1));
        painter.drawRect(barraX, barraY, barraAncho, barraAlto);
    }
}

void Mutante::actualizar()
{
    if (!sistemaVida.estaVivo()) return;

    // Actualizar animación de piernas
    if (animacionSubiendo) {
        frameAnimacion += velocidadAnimacion;
    } else {
        frameAnimacion -= velocidadAnimacion;
    }

    if (frameAnimacion > 3.0f) animacionSubiendo = false;
    if (frameAnimacion < -3.0f) animacionSubiendo = true;

    // Actualizar animación de tentáculos (cada uno con fase diferente)
    anguloTentaculo1 += velocidadTentaculos * 60;
    anguloTentaculo2 += velocidadTentaculos * 60;
    anguloTentaculo3 += velocidadTentaculos * 60;

    if (anguloTentaculo1 >= 360.0f) anguloTentaculo1 -= 360.0f;
    if (anguloTentaculo2 >= 360.0f) anguloTentaculo2 -= 360.0f;
    if (anguloTentaculo3 >= 360.0f) anguloTentaculo3 -= 360.0f;
}

void Mutante::moverHacia(QPointF objetivo)
{
    if (!sistemaVida.estaVivo()) return;

    float centroX = pos_x + ancho / 2.0f;
    float centroY = pos_y + alto / 2.0f;

    float dx = objetivo.x() - centroX;
    float dy = objetivo.y() - centroY;

    float distancia = std::sqrt(dx * dx + dy * dy);

    if (distancia > 0) {
        pos_x += (dx / distancia) * velocidad;
        pos_y += (dy / distancia) * velocidad;
    }
}

void Mutante::mover(float dx, float dy)
{
    pos_x += dx;
    pos_y += dy;
}

void Mutante::recibirDanio(float danio)
{
    sistemaVida.recibirDanio(danio);
}

bool Mutante::estaVivo() const
{
    return sistemaVida.estaVivo();
}
#ifndef ZOMBIEPALA_H

class ZombiePala : public Enemigo

    ~ZombiePala() = default;

    void dibujar(QPainter &painter) override;
    void actualizar() override;
    void moverHacia(QPointF objetivo);

private:
    float frameAnimacion;
    float velocidadAnimacion;
    bool animacionSubiendo;
    float anguloMovimiento;
};

#endif // ZOMBIEPALA_H
#include "zombiepala.h"
#include <cmath>

ZombiePala::ZombiePala(float x, float y)
    : Enemigo(x, y), frameAnimacion(0.0f),
    velocidadAnimacion(0.1f), animacionSubiendo(true), anguloMovimiento(0)
{
    ancho = 35.0f;
    alto = 35.0f;
    velocidad = 2.0f;           // Más rápido que el Cascarudo
    sistemaVida = Vida(80.0f);  // Vida media (Cascarudo=50, Goliat=150)
    danioContacto = 12.0f;      // Daño considerable por la pala
}

void ZombiePala::actualizar() {
    // Animación simple de "caminar"
    if (animacionSubiendo) {
        frameAnimacion += velocidadAnimacion;
        if (frameAnimacion > 3.0f) animacionSubiendo = false;
    } else {
        frameAnimacion -= velocidadAnimacion;
        if (frameAnimacion < -3.0f) animacionSubiendo = true;
    }
}

void ZombiePala::moverHacia(QPointF objetivo) {
    float dx = objetivo.x() - (pos_x + ancho / 2.0f);
    float dy = objetivo.y() - (pos_y + alto / 2.0f);
    float dist = std::sqrt(dx*dx + dy*dy);

    if (dist > 5.0f) {
        anguloMovimiento = std::atan2(dy, dx) * 180.0f / M_PI;
        mover((dx/dist)*velocidad, (dy/dist)*velocidad);
    }
}

void ZombiePala::dibujar(QPainter &p) {
    p.save();
    p.translate(pos_x + ancho/2, pos_y + alto/2);
    p.rotate(anguloMovimiento); // Rotar hacia el jugador

    // Cuerpo (Verde podrido)
    p.setBrush(QColor(60, 100, 60));
    p.setPen(Qt::NoPen);

    // Hombros oscilando al caminar
    float oscilacion = frameAnimacion;

    // Cabeza
    p.drawEllipse(-8, -8, 16, 16);

    // Hombros/Cuerpo
    p.drawRect(-10, -5, 20, 14);

    // Brazos
    p.setBrush(QColor(50, 90, 50));
    // Brazo izquierdo
    p.drawRect(-12, 0 + oscilacion, 4, 10);

    // BRAZO DERECHO CON PALA (Más largo)
    p.save();
    p.rotate(-15); // Un poco inclinado
    p.drawRect(8, -5 - oscilacion, 4, 12); // Brazo

    // Mango de la pala (Marrón madera)
    p.setBrush(QColor(100, 70, 30));
    p.drawRect(10, -20 - oscilacion, 3, 30);

    // Cabeza de la pala (Metal oxidado)
    p.setBrush(QColor(120, 120, 130));
    // Triángulo/Trapecio de la pala
    QPolygonF palaHead;
    palaHead << QPointF(8, -20 - oscilacion)
             << QPointF(15, -20 - oscilacion)
             << QPointF(15, -28 - oscilacion)
             << QPointF(8, -28 - oscilacion);
    p.drawPolygon(palaHead);
    p.restore();

    // Ojos rojos vacíos
    p.setBrush(QColor(200, 50, 50));
    p.drawEllipse(-4, -4, 3, 3);
    p.drawEllipse(4, -4, 3, 3);

    p.restore();
}
